11DESCRIPTION = """Учёт участия студентов в олимпиадах и конкурсах профессионального мастерства. 

12 

13Роли: **student**, **teacher**, **admin**. Гость читает общедоступные данные.

23async def lifespan(app):

32def create_api(version): 

33 api = FastAPI(title="Олимпиадное движение — " + version, version=version, 

34 description=DESCRIPTION, docs_url=None, redoc_url=None, 

35 responses={401: {"description": "Требуется вход"}, 

36 403: {"description": "Недостаточно прав"}, 404: {"description": "Не найдено"}, 

37 409: {"description": "Конфликт данных или состояния"}, 

38 422: {"description": "Ошибка входных данных"}}) 

39 api.include_router(auth.router) 

40 api.include_router(users.router) 

41 if version == "v2": 

42 api.include_router(olympiads.v2_router) 

43 # Общая реализация обеспечивает одинаковые права в обеих версиях. 

44 common_router = APIRouter() 

45 for route in olympiads.router.routes: 

46 if route.path != "/olympiads" or "GET" not in route.methods: 

47 common_router.routes.append(route) 

48 api.include_router(common_router) 

49 else: 

50 api.include_router(olympiads.router) 

51 for router in (participations.router, protocols.router, attachments.router): 

52 api.include_router(router) 

53 

54 @api.get("/docs", include_in_schema=False) 

55 def swagger(): 

56 return get_swagger_ui_html(openapi_url=f"/api/{version}/openapi.json", 

57 title=f"Документация {version}", swagger_js_url="/static/swagger-ui-bundle.js", 

58 swagger_css_url="/static/swagger-ui.css", swagger_favicon_url="data:,") 

59 

60 @api.get("/redoc", include_in_schema=False) 

61 def redoc(): 

62 return get_redoc_html(openapi_url=f"/api/{version}/openapi.json", 

63 title=f"ReDoc {version}", redoc_js_url="/static/redoc.standalone.js", 

64 redoc_favicon_url="data:,", with_google_fonts=False) 

65 return api
